```python
import pandas as pd
import sympy as sp

class EconomicMatrixSystem:
    def __init__(self, variables=None):
        """1. 创建时指定变量"""
        self.variables = variables if variables else []
        # 初始化矩阵，dtype=object 以便放入 sympy 变量
        self.df = pd.DataFrame(
            index=self.variables, 
            columns=self.variables, 
            data=0, 
            dtype=object
        )

    def add_variables(self, new_vars):
            """批量增加新变量"""
            # 过滤掉已经存在的变量
            unique_new_vars = [v for v in new_vars if v not in self.variables]
            if unique_new_vars:
                self.variables.extend(unique_new_vars)
                # 一次性重索引，效率最高
                self.df = self.df.reindex(index=self.variables, columns=self.variables, fill_value=0)
                self.df = self.df.astype(object)

    def set_elasticity(self, row, col, value):
        """4. 填充弹性值（支持实数或 sympy 变量）"""
        self.df.loc[row, col] = value

    def totalelasticity(self):
        """5. 计算总弹性"""
        M = sp.Matrix(self.df)
        I = sp.eye(len(self.variables))
        total_elasticity_matrix = (I - M)**(-1)
        return total_elasticity_matrix
    
    def set_rational_agent(self, rational_var, target1, target2):
        """设定理性变量及其双目标，并自动重组经济矩阵结构"""
        # 1. 创建外生目标变量并设定弹性恒为 1 (Rule 1)
        target1_e = f"{target1}e"
        target2_e = f"{target2}e"
        self.add_variables([target1_e, target2_e])

        # 2. 转移历史依赖：非理性变量 -> 目标的弹性，转移到 -> 外生目标 (Rule 2)
        for v in self.variables:
            if v == rational_var:
                continue # 跳过理性变量本身
            
            # 处理 target1
            val1 = self.df.at[v, target1]
            if val1 != 0:
                self.df.at[v, target1_e] = val1
                self.df.at[v, target1] = 0
            
            # 处理 target2
            val2 = self.df.at[v, target2]
            if val2 != 0:
                self.df.at[v, target2_e] = val2
                self.df.at[v, target2] = 0

        self.df.at[target1_e, target1] = 1
        self.df.at[target2_e, target2] = 1
        
        # 3. 生成外生目标到理性变量的反馈机制 (Rule 3)
        E1 = self.df.at[rational_var, target1]
        E2 = self.df.at[rational_var, target2]
        
        # 逆向弹性计算：原符号为正，新符号为负，反之亦然 (-1 * sign)
        self.df.at[target1_e, rational_var] = -sp.sign(E1) * (1 / (abs(E1)+abs(E2)))
        self.df.at[target2_e, rational_var] = -sp.sign(E2) * (1 / (abs(E1)+abs(E2)))
```

import sympy as sp

α, β, γ,ε = sp.symbols('α β γ ε',positive=True)

model2 = EconomicMatrixSystem(variables=['P','S','D','Se','De'])
model2.set_elasticity('P', 'S', α)
model2.set_elasticity('P', 'D', -β) 
model2.set_elasticity('Se', 'S', 1)
model2.set_elasticity('De', 'D', 1)
model2.set_elasticity('Se','P',-1/(α+β))
model2.set_elasticity('De','P',1/(α+β))
model2.totalelasticity()
