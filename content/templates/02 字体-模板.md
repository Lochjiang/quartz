<%*
// 1. 获取当前在光标处选中的文本
let selectedText = tp.file.selection();

// 如果没有选中文本，弹窗让用户手动输入要修改的文字
if (!selectedText) {
	new Notice("未选择文本！");
	return; // 用户取消输入则终止脚本
}

// 2. 定义常用的预设字体列表（你可以自行增删）
const FONTS = [
    "微软雅黑",
    "黑体",
    "楷体",
    "宋体",
    "仿宋",
    "Arial",
];

// 3. 弹出下拉菜单选择字体
let fontName = await tp.system.suggester(FONTS, FONTS, false, "请选择要应用的字体");

if (!fontName) {
    // 场景 A：用户在下拉菜单按了 ESC 或点击外部，直接终止
    new Notice("已取消字体选择");
    return;
}
// 5. 组合并输出最终代码
// 使用 <span> 标签配合内联 CSS 修改字体，在 Obsidian 中兼容性最好
const result = `<span style="font-family: '${fontName}';">${selectedText}</span>`;

tR += result;
%>
