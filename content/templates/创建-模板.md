<%*
const TYPES = [
  "高级计量经济学",
  "应用统计学",
  "private",
];

/* ⭐ 给枚举加编号（仅用于显示） */
const display = TYPES.map((t, i) => `${i+1}. ${t}`);

/* 下拉选择，false参数为esc退出时不报错，否则抛出错误，默认false*/
let type = await tp.system.suggester(display, TYPES, false, "选择创建文件的位置");

/* ⭐ 如果按ESC -> 直接终止模板 */
if (!type) {
    new Notice("已取消创建"); // 可选提示
    return;
}

/* 自动文件名 */
let newName = tp.date.now("M月D日HH点mm分ss'");

await tp.file.move(`${type}/${newName}`);
%>
