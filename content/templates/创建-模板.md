<%*
const TYPES = [
  "记录",
  "便签",
  "阅读",
  "阅读/古文观止",
];
/* ⭐ 给枚举加编号（仅用于显示） */
const display = TYPES.map((t, i) => `${i+1}. ${t}`);
/* 下拉选择，false参数为esc退出时不报错，否则抛出错误，默认false*/
let type = await tp.system.suggester(display, TYPES, false, "选择创建文件的位置");

/* ⭐ 如果按ESC -> 直接终止模板 */
if (!type) {
    new Notice("已取消创建"); // 可选提示
    return;
}

/* 自动文件名 */
let newName = tp.date.now("M月D日HH点mm分ss'");
let targetPath = `${type}`;

if (type === "记录") {
  let monthFolder = tp.date.now("M月"); // 获取当前月份，例如 "4月"
  targetPath = `${type}/${monthFolder}`; // 拼接为 "记录/4月"
  newName = `记录：${newName}`
  tR += `---\ntags:\n---\n`;
}
if(type === "便签") {
	let monthFolder = tp.date.now("M月"); // 获取当前月份，例如 "4月"
	targetPath = `private/便签/${monthFolder}`; // 拼接为 "记录/4月"
	content = await tp.file.include("[[便签-模板]]");
	if (!content){
	return;}
	tR += content
}

await tp.file.move(`${targetPath}/${newName}`);
%>
