<%*
const TYPES = [
    "词典",
    "文库",
    "健身",
    "Obsidian",
    "自定义",
];

/* 下拉选择 */
let type = await tp.system.suggester(TYPES, TYPES, false, "选择分类");

if (!type) {
    // 场景 A：用户在下拉菜单按了 ESC，直接终止
    new Notice("已取消插入");
    return;
} 

let	targetFolder = `词典/${type}`;

/* ⭐ 1. 使用 if - else if - else 结构处理用户输入 */
if (type === "词典") {
	targetFolder = `词典`; 
}
else if (type === "文库") {
	targetFolder = `文库`; 
}
else if (type === "自定义") {
    // 场景 B：用户选择了“自定义”，弹出输入框
    type = await tp.system.prompt("请输入自定义分类名称：");
    if (!type) {
        new Notice("已取消自定义输入");
        return;
    }
	tR += `---\ntags: ${type}\n---\n`;
	targetFolder = `词典/${type}`;
} 
else {
	tR += `---\ntags: ${type}\n---\n`;
}

/* 重名检测与移动 */
if (app.vault.getAbstractFileByPath(`${targetFolder}/${tp.file.title}.md`)) {
    new Notice(`⚠ 移动失败：${targetFolder}中已存在同名文件`);
} else {
    await tp.file.move(`${targetFolder}/${tp.file.title}`);
    new Notice(`✅ 已成功移动至：${targetFolder}`);
}
%>
