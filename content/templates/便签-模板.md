<%*
// 1. 弹出输入框（第二个参数表示没文字，第三个参数取消时不抛异常，第四个参数 true 表示开启多行文本输入框）
let content = await tp.system.prompt("今天有什么想分享的？", "", false, true);

// 如果用户取消了输入或没有内容，则终止
if (!content) {
    return;
}
// 2. 获取当前时间和设置作者
let time = tp.date.now("YYYY-MM-DD HH:mm");
//let author = ""; // 👈 修改为你的昵称
// 3. 处理文本，给每一行加上 "> " 以符合 Callout 语法
let calloutContent = content.split("\n").map(line => "> " + line).join("\n");
// 拼装最终卡片
let newCard = `> [!weibo] ${time}\n${calloutContent}\n\n`;
// 4. 指定你的动态文件路径（如果在文件夹内，需带上文件夹，如 "生活/动态.md"）
let filePath = "便签墙.md"; 
let file = app.vault.getAbstractFileByPath(filePath);

if (file) {
    // 按照时间倒序插入
    let oldContent = await app.vault.read(file);
	oldContent = oldContent.replace(/^---\n[\s\S]*?\n---\n?/, "");
    let newFileContent = newCard + oldContent;
    await app.vault.modify(file, newFileContent);
} else {
    // 找不到主文件，则创建一个新的
    await app.vault.create(filePath, newCard);
}
tR += content;
new Notice("动态已发布！");
_%>
