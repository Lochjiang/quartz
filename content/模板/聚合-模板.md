<%*
// 1. 弹窗问你要个 Tag（直接输入纯文本，比如 project，不带#）
const tag = await tp.system.prompt("输入要提取的Tag (不带#):");
if (!tag) return;

// 2. 简单粗暴建个同名文件夹（利用 catch 忽略“文件夹已存在”的报错）
await app.vault.createFolder(tag).catch(() => {});

// 3. 把仓库里所有文件翻一遍
for (let file of app.vault.getMarkdownFiles()) {
    
    // 获取文件的 YAML 信息，并强行转换成一段文字
    const cache = app.metadataCache.getFileCache(file);
    const yamlTags = String(cache?.frontmatter?.tags || cache?.frontmatter?.tag || "");
    
    // 只要 YAML 文字里包含你输入的 Tag，并且它不在新文件夹里，就复制！
    if (yamlTags.includes(tag) && !file.path.startsWith(`${tag}/`)) {
        // 利用 catch 忽略“文件已存在”的报错
        await app.vault.copy(file, `${tag}/${file.name}`).catch(() => {});
    }
}

new Notice(`包含 [${tag}] 的文件已复制完毕。`);
%>
