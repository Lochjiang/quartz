<%*
const TYPES = [
	"资料",
];

/* 下拉选择，两个参数，一个是显示，一个是实际*/
let type = await tp.system.suggester(TYPES,TYPES,false,"选择移动的目标文件夹");
/* ⭐ 如果按ESC -> 直接终止模板 */
if (!type) {
    new Notice("已取消移动"); // 可选提示
    return;
}

/* 重名检测 */  
if (app.vault.getAbstractFileByPath(`${type}/${tp.file.title}.md`)) {  
new Notice("⚠目标文件夹中已存在同名文件/本文件");  
} 
else {  
await tp.file.move(`${type}/${tp.file.title}`);  
}
%>
