<%*
// 1. 获取库中所有的文件夹列表
const folders = app.vault.getAllLoadedFiles()
    .filter(x => x instanceof tp.obsidian.TFolder)
    .sort((a, b) => a.path.localeCompare(b.path));

// 2. 使用下拉框 (Suggester) 让用户选择文件夹
// 参数1：显示的文本数组 (文件夹路径)
// 参数2：实际返回的值数组 (文件夹对象)
const selectedFolder = await tp.system.suggester(
    folders.map(f => f.path), 
    folders,
    false, // throw_on_cancel
    "请选择要合并的文件夹" // 提示语
);

// 如果用户按 Esc 取消，则停止执行
if (!selectedFolder) {
    new Notice("已取消合并");
    return;
}

// 3. 获取该文件夹下的所有 Markdown 文件 (不包含子文件夹的内容)
const files = selectedFolder.children
    .filter(f => f instanceof tp.obsidian.TFile && f.extension === "md")
    .sort((a, b) => a.basename.localeCompare(b.basename)); // 按文件名排序

// 4. 循环读取内容并拼接
let mergedContent = "";

if (files.length === 0) {
    mergedContent = "> [!INFO] 提示\n> 所选文件夹内没有 Markdown 文件。";
} else {
    for (const file of files) {
        // 读取文件内容
        let content = await app.vault.read(file);
        // 使用正则表达式将开头的 YAML metadata 替换为空字符串
		content = content.replace(/^---[\s\S]+?---\n/, "");
        // 拼接格式：
        // ## 文件名
        // (空行)
        // 文件内容
        // (空行)
        mergedContent += `## ${file.basename}\n\n${content}\n`;
    }
}

// 5. 将结果输出到当前光标位置
tR += mergedContent;
%>
